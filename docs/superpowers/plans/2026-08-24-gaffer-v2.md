```python
"""Post-assembly EP calibration.

The assembled expected points carry a known level bias (v1 holdout: 60+-minute
starters under-predicted by ~1.1 pts). An isotonic regression per position
group maps assembled ep -> expected actual points. It is fit on out-of-sample
predictions (see train.fit_calibration) so it corrects the pipeline's real
bias, not training-set noise. An unfitted (or absent) model is the identity,
so old model directories keep working.
"""
from __future__ import annotations

import pandas as pd
from sklearn.isotonic import IsotonicRegression

POSITION_GROUPS = ["GKP", "DEF", "MID", "FWD"]
MIN_ROWS = 200


class CalibrationModel:
    def __init__(self) -> None:
        self.by_pos: dict[str, IsotonicRegression] = {}

    def fit(self, ep: pd.Series, actual: pd.Series,
            position: pd.Series) -> "CalibrationModel":
        for pos in POSITION_GROUPS:
            mask = (position == pos).to_numpy()
            if mask.sum() >= MIN_ROWS:
                iso = IsotonicRegression(out_of_bounds="clip")
                iso.fit(ep.to_numpy()[mask], actual.to_numpy()[mask])
                self.by_pos[pos] = iso
        return self

    def apply(self, ep: pd.Series, position: pd.Series) -> pd.Series:
        out = ep.to_numpy(dtype=float).copy()
        for pos, iso in self.by_pos.items():
            mask = (position == pos).to_numpy()
            if mask.any():
                out[mask] = iso.predict(ep.to_numpy()[mask])
        return pd.Series(out, index=ep.index)
```

```python
def fit_calibration(df, tg, scoring) -> "CalibrationModel":
    """Fit the EP calibrator on out-of-sample predictions.

    Components are refit on seasons strictly before the last, then predict the
    last season; isotonic maps (assembled ep -> actual points) on rows with
    minutes > 0. Team-level p_cs/e_gc are held at the backtest's league-average
    constants, so the calibrator corrects the same assembly the replay uses.
    """
    from gaffer.models.calibrate import CalibrationModel
    last = int(df["season_idx"].max())
    train_df = df[df["season_idx"] < last]
    hold = df[df["season_idx"] == last]
    if train_df.empty or hold.empty:
        return CalibrationModel()
    inner = train_all(train_df, tg[tg["season_idx"] < last], save=False,
                      _fit_cal=False)
    comp = predict_components_simple(inner, hold.reset_index(drop=True))
    assembled = assemble_ep(comp, scoring)
    ok = hold.reset_index(drop=True)["minutes"] > 0
    return CalibrationModel().fit(assembled.loc[ok, "ep"],
                                  hold.reset_index(drop=True).loc[ok, "total_points"],
                                  assembled.loc[ok, "position"])
```

- [ ] **Step 2–3: Implement** in `odds.py`:

```python
GOAL_CAP = 10
GRID = [round(0.2 + 0.05 * i, 2) for i in range(int((4.0 - 0.2) / 0.05) + 1)]

def devig(prices: list[float]) -> list[float]:
    implied = [1.0 / p for p in prices]
    s = sum(implied)
    return [x / s for x in implied]

def invert_odds(p_home, p_draw, p_away, p_over25) -> tuple[float, float]:
    """Least-squares grid search for independent-Poisson (mu_h, mu_a)."""
    from math import exp, factorial
    def pmf(mu):
        return [exp(-mu) * mu**k / factorial(k) for k in range(GOAL_CAP + 1)]
    best, best_err = (1.3, 1.3), float("inf")
    for mh in GRID:
        ph_ = pmf(mh)
        for ma in GRID:
            pa_ = pmf(ma)
            win = draw = over = 0.0
            for h in range(GOAL_CAP + 1):
                for a in range(GOAL_CAP + 1):
                    pr = ph_[h] * pa_[a]
                    if h > a: win += pr
                    elif h == a: draw += pr
                    if h + a >= 3: over += pr
            err = ((win - p_home) ** 2 + (draw - p_draw) ** 2
                   + 0.5 * (over - p_over25) ** 2)
            if err < best_err:
                best_err, best = err, (mh, ma)
    return best
```

Performance note: 77×77 grid with an 11×11 inner sum ≈ 700k float ops per fixture — fine for 10 fixtures/week; precompute the pmf list per grid value once outside the double loop (as shown).

```python
"""Rank-aware strategy for the mini-league (spec 2026-08-24 §3).

lam = sign * LAMBDA_CAP * clamp(|gap| / (2*SIGMA*sqrt(W)) - 0.5, 0, 1)
Positive lam chases (favor differentials), negative defends (mirror rivals),
zero leaves the optimizer exactly at v1 points-max.
"""
from __future__ import annotations

import math
from dataclasses import dataclass

import pandas as pd

SIGMA = 18.0        # per-GW score sigma, pinned; later: estimate from tracking
LAMBDA_CAP = 0.5
LAST_GW = 38


@dataclass
class Strategy:
    lam: float
    gap: int
    weeks_left: int
    stance: str          # "chase" | "defend" | "neutral"
    rival_name: str


def compute_strategy(my_total: int, rivals: pd.DataFrame,
                     current_gw: int) -> Strategy:
    weeks = max(1, LAST_GW - current_gw + 1)
    if rivals.empty:
        return Strategy(0.0, 0, weeks, "neutral", "")
    top = rivals.sort_values("total", ascending=False).iloc[0]
    if int(top["total"]) > my_total:            # chasing the leader
        gap, sign, rival = int(top["total"]) - my_total, +1, str(top["entry_name"])
    else:                                       # leading: defend vs top rival
        gap, sign, rival = my_total - int(top["total"]), -1, str(top["entry_name"])
    raw = gap / (2 * SIGMA * math.sqrt(weeks)) - 0.5
    lam = sign * LAMBDA_CAP * min(max(raw, 0.0), 1.0)
    stance = "neutral" if lam == 0 else ("chase" if lam > 0 else "defend")
    return Strategy(lam, gap, weeks, stance, rival)


def win_probability(my_total: int, their_total: int, weeks_left: int) -> float:
    """P(I finish above them): normal approximation, independent scores."""
    if weeks_left <= 0:
        return 1.0 if my_total >= their_total else 0.0
    z = (my_total - their_total) / (SIGMA * math.sqrt(2 * weeks_left))
    return 0.5 * (1 + math.erf(z / math.sqrt(2)))
```

```python
def tilt_ep(ep_by: dict, eo_pct: dict, lam: float) -> dict:
    """Tilted EP for pool construction ONLY. Raw ep is reported everywhere.

    eo_pct: league effective ownership in percent (may exceed 100 via
    captaincy); clamped to [0, 1] as a fraction for the tilt.
    """
    if lam == 0.0:
        return dict(ep_by)
    out = {}
    for key, ep in ep_by.items():
        code = key[0]
        eo1 = min(eo_pct.get(code, 0.0) / 100.0, 1.0)
        out[key] = ep * (1 + lam * (1 - eo1))
    return out
```
